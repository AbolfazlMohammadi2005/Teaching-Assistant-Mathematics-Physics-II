	\section{سوال اول}
	محاسبه کنید
	\[
	\int_0^{2\pi} \frac{d\theta}{a \pm b \cos \theta}.
	\]  
	چه اتفاقی می‌افتد اگر \(|b| > |a|\)؟
	
	همچنین محاسبه کنید
	\[
	\int_0^{2\pi} \frac{d\theta}{a \pm b \sin \theta} = \frac{2\pi}{\sqrt{a^2 - b^2}}, \quad a > |b|.
	\]
	\section{سوال دوم}
	نشان دهید که
	\[
	\int_0^{2\pi} \frac{d\theta}{(a + \cos\theta)^2} = \frac{\pi a}{(a^2 - 1)^{3/2}}, \quad a > 1.
	\]
	
	
	\section{سوال سوم}
	محاسبه کنید
	\[
	\int_0^{2\pi} \frac{d\theta}{1 - 2t \cos\theta + t^2} = \frac{2\pi}{1 - t^2}, \quad |t| < 1.
	\]  
	چه اتفاقی می‌افتد اگر \(|t| > 1\) یا \(|t| = 1\)؟
	
	\section{سوال چهارم}
	
	با استفاده از حساب مانده‌ها نشان دهید که
	\[
	\int_0^\pi \cos^{2n} \theta\, d\theta = \pi \frac{(2n-1)!!}{(2n)!!} = \frac{\pi (2n-1)!!}{2^{2n} (n!)^2}, \quad n = 0,1,2,\dots
	\]  
	\textit{راهنمایی: از رابطه \(\cos \theta = \frac{1}{2}(e^{i\theta} + e^{-i\theta}) = \frac{1}{2}(z + z^{-1}), \; |z|=1\) استفاده کنید.}
	\section{سوال پنجم}
	انتگرال
	\[
	\int_{-\infty}^{\infty} \frac{\cos bx - \cos ax}{x^2}\, dx, \quad a > b > 0
	\]  
	را محاسبه کنید.  
	
	\section{سوال ششم}
	اثبات کنید که
	\[
	\int_{-\infty}^{\infty} \frac{\sin^2 x}{x^2}\, dx = \pi.
	\]  
	\textit{راهنما: از رابطه \(\sin x = \frac{1}{2}(1 - \cos 2x)\) استفاده کنید.}
	
	
	\section{سوال هفتم}
	نشان دهید که
	\[
	\int_0^\pi \frac{x \sin x}{x^2 + 1}\, dx = \frac{\pi}{2e}.
	\]
	
	\section{سوال هشتم}
نشان دهید که برای \(a>0\):
\[
(a)\quad \int_{-\infty}^{\infty} \frac{\cos x}{x^2 + a^2}\, dx = \frac{\pi}{a} e^{-a}.
\]  
اگر \(\cos x\) با \(\cos kx\) جایگزین شود، سمت راست چگونه تغییر می‌کند؟

\[
(b)\quad \int_{-\infty}^{\infty} \frac{x \sin x}{x^2 + a^2}\, dx = \pi e^{-a}.
\]  
اگر \(\sin x\) با \(\sin kx\) جایگزین شود، سمت راست چگونه تغییر می‌کند؟
	
	\section{سوال نهم}
	
	یک محاسبه مکانیک کوانتومی برای احتمال گذار منجر به تابع
	\[
	f(t,\omega) = \frac{2(1 - \cos \omega t)}{\omega^2}
	\]  
	می‌شود. نشان دهید که
	\[
	\int_{-\infty}^{\infty} f(t,\omega)\, d\omega = 2\pi t.
	\]
