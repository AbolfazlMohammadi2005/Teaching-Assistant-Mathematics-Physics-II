	\section{سوال اول}
	نشان دهید که
	\[
	\int_{z_1}^{z_2} f(z)\, dz = - \int_{z_2}^{z_1} f(z)\, dz.
	\]
	
	\section{سوال دوم}
	تاثبات کنید که
	\[
	\left| \int_C f(z)\, dz \right| \le |f|_{\text{\lr{max}}} \cdot L,
	\] 
	که در آن \(|f|_{\text{\lr{max}}}\) بزرگ‌ترین مقدار \(|f(z)|\) در طول مسیر \(C\) و \(L\) طول مسیر است.
	
	
	\section{سوال سوم}
	نشان دهید که انتگرال
	\[
	\int_{3+4i}^{4-3i} (4z^2 - 3i z)\, dz
	\] 
	روی دو مسیر زیر یک مقدار دارد:
	\\  
	\textbf{(الف)} خط مستقیم بین نقاط ابتدا و انتها  
	\\
	\textbf{(ب)} قوس دایره‌ای با شعاع \(|z| = 5\).
	\\
	\section{سوال چهارم}
	اگر 
	\[
	F(z) = \int_{z_0}^{z} f(z)\, dz,
	\]  
	نشان دهید که \(F(z)\) به مسیر بین حدود انتگرال وابسته نیست و مقدار \(F(\pi i)\) را محاسبه کنید.
	
	\section{سوال پنجم}
	انتگرال
	\[
	\int_C (x^2 - i y^2)\, dz
	\]  
	را محاسبه کنید، که مسیر انتگرال‌گیری به صورت:  
	\\
	\textbf{(الف)} حول دایره واحد در جهت ساعتگرد  
	\\
	\textbf{(ب)} روی مربعی با رئوس \(\pm 1 \pm i\) است.  
	\\
	توضیح دهید چرا نتایج قسمت‌های (الف) و (ب) یکسان یا متفاوت هستند.
	
	\section{سوال ششم}
	اثبات کنید که
	\[
	\oint_C \frac{dz}{z^2 + z} = 0,
	\]  
	که مسیر \(C\) دایره‌ای با \(|z| = R > 1\) است. 
	\\ 
	\textit{راهنما: استفاده مستقیم از قضیه انتگرال کوشی مجاز نیست. انتگرال را با تجزیه به کسرهای جزئی محاسبه کنید؛ این مقدار برای \(R>1\) صفر و برای \(R<1\) برابر \(2\pi i\) می‌شود.}
