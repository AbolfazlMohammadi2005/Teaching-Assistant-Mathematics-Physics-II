	\section{سوال اول}
	نشان دهید که آیا تابع \( f(z) = \mathrm{Re}(z) = x \) تحلیلی است یا خیر.
	
	\section{سوال دوم}
	 تابع تحلیلی \( w(z) = u(x,y) + iv(x,y) \) را بیابید:  
	 \\
	\textbf{(الف)} اگر \( u(x,y) = x^3 - 3xy^2 \)،
	\\  
	\textbf{(ب)} اگر \( v(x,y) = e^{-y} \sin x \).
	
	\section{سوال سوم}
	اگر منطقه مشترکی وجود داشته باشد که در آن \( w_1 = u(x,y) + iv(x,y) \) و \( w_2 = w_1^* = u(x,y) - iv(x,y) \) هر دو تحلیلی باشند، نشان دهید که \( u(x,y) \) و \( v(x,y) \) ثابت هستند.
	
	\section{سوال چهارم}
	با شروع از \( f(z) = \frac{1}{x + iy} \)، نشان دهید که \( 1/z \) در کل صفحه مختلط متناهی به جز نقطه \( z = 0 \) تحلیلی است. این مسئله بحث ما درباره تحلیلی بودن \( z^n \) را برای توان‌های صحیح منفی \( n \) گسترش می‌دهد.
	\section{سوال پنجم}
	نشان دهید که با توجه به معادلات کوشی-ریمان، مشتق \( f'(z) \) برای \( dz = a\, dx + i b\, dy \) (با \( a \neq 0 \) و \( b \neq 0 \)) همان مقداری را دارد که برای \( dz = dx \) دارد.
	
	\section{سوال ششم}
	با استفاده از \( f(r e^{i\theta}) = R(r,\theta)e^{i\Theta(r,\theta)} \)، که در آن \( R(r,\theta) \) و \( \Theta(r,\theta) \) توابع واقعی قابل مشتق‌گیری از \( r \) و \( \theta \) هستند، نشان دهید که شرایط کوشی-ریمان در مختصات قطبی به صورت زیر درمی‌آید:  
	\\
	\textbf{(الف)} \( \frac{\partial R}{\partial r} = \frac{R}{r} \frac{\partial \Theta}{\partial \theta} \)  
	\\
	\textbf{(ب)} \( \frac{1}{r}\frac{\partial R}{\partial \theta} = -R \frac{\partial \Theta}{\partial r} \)  
	\\
	\textit{راهنمایی: ابتدا مشتق را با \(\delta z\) شعاعی و سپس با \(\delta z\) مماسی تنظیم کنید.}
	
	
	\section{سوال هفتم}
	برای هر یک از توابع زیر \( f(z) \)، \( f'(z) \) را بیابید و بزرگ‌ترین ناحیه‌ای که در آن \( f(z) \) تحلیلی است مشخص کنید: 
	\\ 
	\textbf{(الف)} \( f(z) = \sin z \)  
	\\
	\textbf{(ب)} \( f(z) = \frac{1}{z} \)  
	\\
	\textbf{(پ)} \( f(z) = \frac{1}{z+1} \) 
	\\ 
	\textbf{(ت)} \( f(z) = \frac{1}{z(z+1)} \) 
	\\ 
	\textbf{(ث)} \( f(z) = e^{-1/z} \)  
	\\
	\textbf{(ج)} \( f(z) = z^2 - 3z + 2 \)  
	\\
	\textbf{(چ)} \( f(z) = \tan(z) \)  
	\\
	\textbf{(ح)} \( f(z) = \tanh(z) \)
	\section{سوال هشتم}
	 برای چه مقادیر مختلط، هر یک از توابع زیر مشتق دارند؟  
	\textbf{(الف)} \( f(z) = z^{3/2} \)  
	\\
	\textbf{(ب)} \( f(z) = z^{-3/2} \)  
	\\
	\textbf{(پ)} \( f(z) = \tan^{-1}(z) \) 
	\\ 
	\textbf{(ت)} \( f(z) = \tanh^{-1}(z) \)

	\section{سوال نهم}
	 جریان سیال دو بعدی غیرچرخشی با پتانسیل مختلط \( f(z) = u(x,y) + i v(x,y) \) توصیف می‌شود. قسمت حقیقی \( u(x,y) \) پتانسیل سرعت و قسمت موهومی \( v(x,y) \) تابع جریان است. سرعت سیال \( \mathbf{V} = \nabla u \) است. اگر \( f(z) \) تحلیلی باشد:  
	 \\
	\textbf{(الف)} نشان دهید که \( \frac{df}{dz} = V_x - i V_y \)  
	\\
	\textbf{(ب)} نشان دهید که \( \nabla \cdot \mathbf{V} = 0 \) (منبع یا چاه ندارد)  
	\\
	\textbf{(پ)} نشان دهید که \( \nabla \times \mathbf{V} = 0 \) (جریان غیرچرخشی و بدون آشفتگی است)
	\\
