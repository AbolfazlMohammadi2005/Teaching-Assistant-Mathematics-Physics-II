	\section{سوال اول}
	نشان دهید که برای \(a>0\):
	\[
	\int_{-\infty}^{\infty} \frac{dx}{(x^2 + a^2)^2} = \frac{\pi}{4 a^3}.
	\]
	\section{سوال دوم}
	\[
	\int_{-\infty}^{\infty} \frac{dx}{x^2 \sqrt{1+x^4}} = \frac{\pi}{2}.
	\]  
	
	همچنین
	\[
	\int_0^{\infty} \frac{x^p \ln x}{x^2 + 1} \, dx = \frac{\pi}{4} \frac{\sin(\pi p/2)}{\cos^2(\pi p/2)}, \quad 0 < p < 1.
	\]
	
	\section{سوال سوم}
	محاسبه کنید:
	\[
	\int_0^{\infty} \frac{(\ln x)^2}{1+x^2} \, dx
	\]  
	\textbf{(الف)} با استفاده از بسط سری مناسب مخرج:
	\[
	\sum_{n=0}^{\infty} \frac{(-1)^n}{(2n+1)^3}
	\]  
	\textbf{(ب)} و با استفاده از انتگرال‌گیری روی کانتور برای به دست آوردن:
	\[
	\frac{\pi^3}{8}.
	\]
	
	\section{سوال چهارم}
	اثبات کنید که
	\[
	\int_0^\infty \frac{\ln(1+x^2)}{1+x^2}\, dx = \pi \ln 2.
	\]
	\section{سوال پنجم}
	نشان دهید که برای \(-1 < a < 1\):
	\[
	\int_0^\infty \frac{x^a}{x+1}\, dx = \frac{\pi a}{\sin \pi a}.
	\]  
	
	\section{سوال ششم}
	نشان دهید که
	\[
	\int_{-\infty}^\infty \frac{x^2 \, dx}{x^4 - 2x^2 \cos 2\theta + 1} = \frac{\pi}{2} \frac{\sin \theta}{(1 - \cos 2\theta)^{1/2}}.
	\]  
	
	\section{سوال هفتم}
	نشان دهید که
	\[
	\int_0^{\infty} \frac{dx}{1+x^n} = \frac{\pi/n}{\sin(\pi/n)}.
	\]  
	\section{سوال هشتم}
	\textbf{(الف)} نشان دهید که
	\[
	f(z) = z^4 - 2 z^2 \cos 2\theta + 1
	\]  
	دارای صفرهای
	\[
	e^{i\theta}, \; e^{-i\theta}, \; -e^{i\theta}, \; -e^{-i\theta}
	\]  
	است.  
	
	\textbf{(ب)} نشان دهید که
	\[
	\int_{-\infty}^\infty \frac{dx}{x^4 - 2x^2 \cos 2\theta + 1} = \frac{\pi}{2} \frac{\sin \theta}{(1 - \cos 2\theta)^{1/2}}.
	\]  
	\section{سوال نهم}
	
	نشان دهید که برای \(0 < a < 1\):
	\[
	\int_0^\infty \frac{x^{-a}}{x+1}\, dx = \frac{\pi}{\sin(\pi a)}.
	\]  
	\section{سوال دهم}
	
	نشان دهید که برای \(|b| < 1\):
	\[
	\int_0^\infty \frac{\cosh(bx)}{\cosh x}\, dx = \frac{\pi}{2 \cos(\pi b/2)}.
	\]  
	\section{سوال یازدهم}
	نشان دهید که:
	\[
	\int_0^\infty \cos(t^2)\, dt = \int_0^\infty \sin(t^2)\, dt = \frac{\sqrt{\pi}}{2}.
	\]  
	\section{سوال دوازدهم}
	نشان دهید که:
	\[
	\int_0^1 \frac{dx}{(x^2 - x^3)^{1/3}} = \frac{2\pi}{3}.
	\]  
	\section{سوال سیزدهم}
	محاسبه کنید:
	\[
	\int_{-\infty}^\infty \frac{\tan^{-1}(ax)}{x(x^2 + b^2)}\, dx,
	\]  
	برای \(a, b > 0\) و \(ab < 1\).  
	توضیح دهید که چرا انتگرال در \(x = 0\) ‌قطبی ندارد.
