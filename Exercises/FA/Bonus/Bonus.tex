	\section{سوال اول}
	تحلیلی بودن توابع زیر را بررسی کنید.
	
	\begin{enumerate}
		\item \( f(z) = \mathrm{Re}(z) \)
		\item \( f(z) = z \, \mathrm{Im}(z) \)
		\item \( f(z) = z \, \mathrm{Im}(z^2) \)
		\item \( f(z) = e^{z} \)
		\item \( f(z) = \cos z \)
		\item \( f(z) = \cosh z \)
		\item \( f(z) = z^8 \)
		\item \( f(z) = \dfrac{z}{z+1} \)
		\item \( f(z) = z\,\mathrm{Re}(z + i z^2) \)
	\end{enumerate}
	\section{سوال دوم}
	همساز بودن توابع زیر را بررسی کنید.  
	در صورت همساز بودن، مزدوج همساز آن‌ها را بیابید.
	
	\begin{enumerate}
		\item \( u = x^2 - y^2 \)
		\item \( u = \mathrm{Re}(\sin z) \)
		\item \( v = \ln|z| \)
	\end{enumerate}
	
	\textbf{مثال ۴:}  
	مقادیر \(\alpha\) و \(\beta\) را طوری بیابید که تابع  
	\( u = e^{\alpha x} \cos(\beta y) \)  
	همساز باشد.
	
	\vspace{8pt}
	\textbf{مثال ۵:}  
	 تابع  
	\( u = e^{a x+byi} \)  
	به ازای چه مقادیری از \( a \) و \( b \) همساز است؟
	
	\section{سوال سوم}
	سری لوران توابع زیر را بیابید:
	
	\begin{enumerate}
		\item \( f(z) = \dfrac{\sin(z)}{z^2} \)
		\item \( f(z) = \dfrac{e^{1/z}}{z - 1} \)
		\item \( f(z) = \dfrac{1 - \cos(z}{z^2} \)
		\item \( f(z) = \dfrac{\cos(z)}{z^2} \)
		\item \( f(z) = \dfrac{\ln(1+z)}{z^3} \)
		\item \( f(z) = \dfrac{1}{z\ln (1+z)} \)
		
	\end{enumerate}
	\section{سوال چهارم}
	
	حاصل انتگرال زیر را محاسبه کنید:
	
	\[
	\oint_{C} z^2 \, dz
	\]
	
	که در آن مسیر \( C \) دارای معادله‌ی پارامتری زیر است:
	
	\[
	\begin{cases}
		z = t + i t^2 \\
		0 \le t \le 1
	\end{cases}
	\]
	\section{سوال پنجم}
	
	حاصل انتگرال زیر را بیابید:
	
	\[
	\int_{0}^{1+i} {(ze^{z}+ |z|+e^{z})dz}
	\]
	
	که در آن مسیر انتگرال، خط \( y = x \) می‌باشد.
	\section{سوال هفتم}
	
	اگر \( C \) پاره‌خط واصل از نقطه‌ی \( i \) به نقطه‌ی \( 1 \) باشد،  
	حاصل انتگرال زیر را بیابید:
	
	\[
	\oint_{C} z \, z^{*} \, d z^{*}
	\]
